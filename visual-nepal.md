<script>
(function () {
  function fit(box) {
    var iframe = box.querySelector('iframe');
    if (!iframe) return;
    var nativeW = parseInt(box.dataset.nativeW, 10) || 1440;
    var nativeH = parseInt(box.dataset.nativeH, 10) || 900;
    var cropX = parseInt(box.dataset.cropX, 10) || 0;
    var visibleW = nativeW - cropX;
    var scale = box.clientWidth / visibleW;
    iframe.style.width = nativeW + 'px';
    iframe.style.height = nativeH + 'px';
    iframe.style.transform = 'scale(' + scale + ')';
    iframe.style.left = (-cropX * scale) + 'px';
  }
  function fitAll() {
    document.querySelectorAll('.visual-card-preview').forEach(fit);
  }
  fitAll();
  var t;
  window.addEventListener('resize', function () {
    clearTimeout(t);
    t = setTimeout(fitAll, 150);
  });
})();
</script>

<script>
(function () {
  function openShareUrl(url) {
    window.open(url, '_blank', 'noopener,width=600,height=520');
  }
  document.querySelectorAll('.visual-card').forEach(function (card) {
    var slug = card.dataset.shareSlug;
    var title = card.dataset.shareTitle;
    if (!slug) return;
    // Points at a pre-generated share page carrying Open Graph tags for a
    // real branded PNG, so Twitter/Facebook show the image automatically.
    var shareUrl = 'https://8848.live/share/visual-nepal/' + slug + '/';
    var text = title + ' — via 8848.live';
    var row = card.querySelector('.visual-card-share-row');
    if (!row) return;
    row.addEventListener('click', function (e) {
      var btn = e.target.closest('.share-btn');
      if (!btn) return;
      var kind = btn.getAttribute('data-share');
      if (kind === 'twitter') {
        openShareUrl('https://twitter.com/intent/tweet?text=' + encodeURIComponent(text) + '&url=' + encodeURIComponent(shareUrl));
      } else if (kind === 'facebook') {
        openShareUrl('https://www.facebook.com/sharer/sharer.php?u=' + encodeURIComponent(shareUrl));
      } else if (kind === 'instagram') {
        // Instagram has no link-preview share intent -- open the branded
        // share image directly so it can be saved/shared from there.
        window.open('https://8848.live/assets/shares/visual-nepal/' + slug + '.png', '_blank', 'noopener');
      }
    });
  });
})();
</script>
